cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75)
endif()
project(nested_dissection_gpu LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_library(ndgpu src/ordering.cu)
target_include_directories(ndgpu PUBLIC include)
target_link_libraries(ndgpu PUBLIC CUDA::cudart)
target_compile_features(ndgpu PUBLIC cxx_std_17 cuda_std_17)

include(CTest)
if(BUILD_TESTING)
    add_executable(ndgpu_test tests/ordering_test.cu)
    target_link_libraries(ndgpu_test PRIVATE ndgpu)
    add_test(NAME ordering COMMAND ndgpu_test)
    add_test(NAME reference_grids COMMAND ndgpu_test --reference-grids)
endif()
add_executable(ndgpu_grid examples/grid.cu)
target_link_libraries(ndgpu_grid PRIVATE ndgpu)

option(NDGPU_WITH_CUDSS "Build the cuDSS handoff/solve example" OFF)
if(NDGPU_WITH_CUDSS)
    find_path(CUDSS_INCLUDE_DIR cudss.h HINTS ${CUDSS_ROOT}/include REQUIRED)
    find_library(CUDSS_LIBRARY NAMES cudss libcudss.so.0 HINTS ${CUDSS_ROOT}/lib REQUIRED)
    add_executable(ndgpu_cudss examples/cudss.cu)
    target_include_directories(ndgpu_cudss PRIVATE ${CUDSS_INCLUDE_DIR})
    target_link_libraries(ndgpu_cudss PRIVATE ndgpu ${CUDSS_LIBRARY})
    if(BUILD_TESTING)
        add_test(NAME cudss_handoff COMMAND ndgpu_cudss 8)
    endif()
endif()

option(NDGPU_BUILD_BENCHMARKS "Build the SuiteSparse/cuDSS ordering comparison" OFF)
if(NDGPU_BUILD_BENCHMARKS)
    if(NOT NDGPU_WITH_CUDSS)
        message(FATAL_ERROR "NDGPU_BUILD_BENCHMARKS requires NDGPU_WITH_CUDSS=ON")
    endif()
    find_path(CXSPARSE_INCLUDE_DIR cs.h PATH_SUFFIXES suitesparse REQUIRED)
    find_library(CXSPARSE_LIBRARY cxsparse REQUIRED)
    add_executable(ndgpu_compare benchmarks/compare.cu)
    target_include_directories(ndgpu_compare PRIVATE ${CUDSS_INCLUDE_DIR} ${CXSPARSE_INCLUDE_DIR})
    target_link_libraries(ndgpu_compare PRIVATE ndgpu ${CUDSS_LIBRARY} ${CXSPARSE_LIBRARY})
    if(BUILD_TESTING)
        add_executable(ndgpu_benchmark_test tests/benchmark_test.cu)
        target_include_directories(ndgpu_benchmark_test PRIVATE ${CXSPARSE_INCLUDE_DIR})
        target_link_libraries(ndgpu_benchmark_test PRIVATE ndgpu ${CXSPARSE_LIBRARY})
        add_test(NAME benchmark_input_and_counts COMMAND ndgpu_benchmark_test)
    endif()
endif()
